        float time;
        float2 resolution;
        float2 mouse;
    };

Wavey Sine Shader Example
---------------------------------------------------------------------------------------------

    local shaderObject = Drawing.new("Shader")
    
    shaderObject.Vertex = [[\
    struct VSOut\
    {\
        float4 pos : SV_POSITION;\
        float2 uv  : TEXCOORD0;\
    };\
    \
    VSOut main(uint id : SV_VertexID)\
    {\
        VSOut o;\
    \
        float2 pos[3] = {\
            float2(-1.0, -1.0),\
            float2(-1.0,  3.0),\
            float2( 3.0, -1.0)\
        };\
    \
        float2 uv[3] = {\
            float2(0.0, 1.0),\
            float2(0.0, -1.0),\
            float2(2.0, 1.0)\
        };\
    \
        o.pos = float4(pos[id], 0.0, 1.0);\
        o.uv  = uv[id];\
    \
        return o;\
    }]]
    
    shaderObject.Pixel = [[\
    cbuffer Params : register(b0)\
    {\
        float time;\
    	float2 resolution;\
        float2 mouse;\
    };\
    \
    float4 main(float4 pos : SV_Position, float2 uv : TEXCOORD0) : SV_Target\
    {\
        float2 p = uv - 0.5;\
        p *= 2.0;\
    \
        for (int i = 1; i < 5; i++)\
        {\
            float fi = (float)(i + 11);\
    \
            float2 freq  = float2(1.6, 1.1) * fi;\
            float2 phase = time * (float)i * float2(3.4, 0.5) / 10.0;\
    \
            p += sin(p.yx * freq + phase) * 0.1;\
        }\
    \
        float c = abs(sin(p.y) + sin(p.x)) * 0.5;\
    \
        return float4(c.xxx, 1.0);\
    }]]
    
    shaderObject:Create()
