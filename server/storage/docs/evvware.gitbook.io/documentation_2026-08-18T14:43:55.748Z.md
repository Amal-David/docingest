### Minimal callback setup[](https://evvware.gitbook.io/evvware-api/getting-started/basics#minimal-callback-setup)

Most scripts start by registering one or more callbacks.

Copy

    local function on_create_move(cmd)
        if not cmd then return end
    end
    
    local function on_paint()
    end
    
    client.register_callback("create_move", on_create_move)
    client.register_callback("paint_traverse", on_paint)

### Print to console and show notifications[](https://evvware.gitbook.io/evvware-api/getting-started/basics#print-to-console-and-show-notifications)

Use `client` for quick feedback while testing.

Copy

    client.print("hello from lua")
    client.print("colored message", color.new(120, 255, 120, 255))
    
    client.add_notify(
        "script loaded",
        true,
        true,
        color.new(255, 255, 255, 255),
        4.0
    )

### Read the local player[](https://evvware.gitbook.io/evvware-api/getting-started/basics#read-the-local-player)

Use `entities` to get the local player, then read values from `cs_player`.

Copy

    local function on_create_move(cmd)
        local local_player = entities.get_local_player()
        if not local_player then return end
        if not local_player:is_alive() then return end
    
        local health = local_player:get_health()
        local origin = local_player:origin()
        local velocity = local_player:velocity()
    
        client.print("health: " .. tostring(health))
        client.print("origin: " .. tostring(origin))
        client.print("velocity: " .. tostring(velocity))
    end

### Read engine state[](https://evvware.gitbook.io/evvware-api/getting-started/basics#read-engine-state)

Use `engine` for map, connection, screen, and view data.

Copy

    local screen = engine.get_screen_size()
    local view = engine.get_view_angles()
    
    client.print("map: " .. tostring(engine.get_map_name()))
    client.print("screen: " .. tostring(screen))
    client.print("view: " .. tostring(view))
    client.print("in game: " .. tostring(engine.is_in_game()))

### Loop players[](https://evvware.gitbook.io/evvware-api/getting-started/basics#loop-players)

Use `engine.get_active_players_data()` when you need cached player data.

Copy

    for _, player in ipairs(engine.get_active_players_data()) do
        client.print(
            "index: " .. tostring(player.index) ..
            " alive: " .. tostring(player.alive) ..
            " teammate: " .. tostring(player.teammate)
        )
    end

### Handle game events[](https://evvware.gitbook.io/evvware-api/getting-started/basics#handle-game-events)

Register the event name first, then handle it in `game_event_listener`.

Copy

    local function on_game_event(event)
        local name = tostring(event:get_name())
    
        if name == "bullet_impact" then
            client.print("bullet impact")
        end
    
        if name == "player_hurt" then
            local userid = event:get_int("userid")
            client.print("hurt userid: " .. tostring(userid))
        end
    end
    
    -- also, you can register more events
    -- client.register_event("bomb_planted")
    -- check out: 
    -- wiki.alliedmods.net/Counter-Strike:_Source_Events
    -- wiki.alliedmods.net/Generic_Source_Events
    -- make sure game supports event you choosed
    
    client.register_callback("game_event_listener", on_game_event)

### Draw on screen[](https://evvware.gitbook.io/evvware-api/getting-started/basics#draw-on-screen)

Use `render` inside `paint_traverse`.

Copy

    local tahoma = render.create_font("Tahoma", 12, 400, 0, 0, 16)
    
    local function on_paint()
        render.filled_rect(20, 20, 220, 60, color.new(20, 20, 20, 180))
        render.outlined_rect(20, 20, 220, 60, color.new(255, 255, 255, 255))
    
        render.text(
            tahoma.font,
            text_align_left,
            false,
            false,
            30,
            40,
            color.new(255, 255, 255, 255),
            "Hello from render"
        )
    end

### Convert world to screen[](https://evvware.gitbook.io/evvware-api/getting-started/basics#convert-world-to-screen)

Use `world2screen` before drawing world positions.

Copy

    local function on_paint()
        local local_player = entities.get_local_player()
        if not local_player then return end
    
        local screen_pos = render.world2screen(local_player:origin())
        if screen_pos.x == -1 and screen_pos.y == -1 then return end
    
        render.filled_rect(
            screen_pos.x - 2,
            screen_pos.y - 2,
            4,
            4,
            color.new(255, 0, 0, 255)
        )
    end

### Build a menu[](https://evvware.gitbook.io/evvware-api/getting-started/basics#build-a-menu)

Use `create_ui_manager()` and add widgets to a frame.

Copy

    local ui_manager = create_ui_manager()
    local frame = ui_manager:add_frame()
    
    frame:show()
    frame:text("Example script")
    frame:jump()
    
    local enabled = frame:add_checkbox("Enable", false, function(value)
        client.print("enabled: " .. tostring(value))
    end)
    
    local size = frame:add_slider_int("Size", 10, 1, 50, function(value)
        client.print("size: " .. tostring(value))
    end)

### Read widget values[](https://evvware.gitbook.io/evvware-api/getting-started/basics#read-widget-values)

Most menu widgets return a `menu_widget`.

Copy

    local function on_paint()
        if not enabled:get_bool() then return end
    
        local box_size = size:get_int()
        render.filled_rect(300, 200, box_size, box_size, color.new(255, 255, 255, 255))
    end

### Work with vectors and colors[](https://evvware.gitbook.io/evvware-api/getting-started/basics#work-with-vectors-and-colors)

`vector2`, `vector3`, and `color` are used everywhere.

Copy

    local a = vector3.new(10, 0, 0)
    local b = vector3.new(5, 5, 0)
    local c = a + b
    
    client.print("vector result: " .. tostring(c))
    client.print("length: " .. tostring(c:length()))
    
    local white = color.new(255, 255, 255, 255)
    client.print("color alpha: " .. tostring(white.a))

### Trace a ray[](https://evvware.gitbook.io/evvware-api/getting-started/basics#trace-a-ray)

Use `game_ray`, `game_trace`, and a trace filter.

Copy

    local function on_create_move(cmd)
        local local_player = entities.get_local_player()
        if not local_player then return end
    
        local start_pos = local_player:eye_position()
        local end_pos = start_pos + vector3.new(100, 0, 0)
    
        local ray = game_ray.new()
        ray:init(start_pos, end_pos)
    
        local trace = game_trace.new()
        local filter = trace_filter_custom.new(function(ent, mask)
            return true
        end)
    
        engine.trace_ray(ray, MASK_SOLID, filter, trace)
    
        client.print("fraction: " .. tostring(trace.fraction))
    end

### Use hitbox helpers[](https://evvware.gitbook.io/evvware-api/getting-started/basics#use-hitbox-helpers)

        local box_min, box_max = client.get_hitbox_box(local_player, 12)
    
        client.print("hitbox pos: " .. tostring(hitbox_pos))
        client.print("box min: " .. tostring(box_min))
        client.print("box max: " .. tostring(box_max))
    end

### Add custom ESP flags[](https://evvware.gitbook.io/evvware-api/getting-started/basics#add-custom-esp-flags)

### Get the active weapon config path[](https://evvware.gitbook.io/evvware-api/getting-started/basics#get-the-active-weapon-config-path)

Use the local player weapon with `config.get_current_aimbot_config_path()`.

Copy

    local function on_create_move(cmd)
        local local_player = entities.get_local_player()
        if not local_player then return end
    
        local weapon = local_player:get_active_weapon()
        if not weapon then return end
    
        local path = config.get_current_aimbot_config_path(weapon)
        client.print("config path: " .. tostring(path))
    end

### Full starter example[](https://evvware.gitbook.io/evvware-api/getting-started/basics#full-starter-example)

This combines callbacks, player lookup, render, and menu state.

Copy

    local ui_manager = create_ui_manager()
    local frame = ui_manager:add_frame()
    frame:show()
    
    local enabled = frame:add_checkbox("Draw local player box", true)
    local font = render.create_font("Tahoma", 12, 400, 0, 0, 16)
    
    local function on_paint()
        if not enabled:get_bool() then return end
    
        local local_player = entities.get_local_player()
        if not local_player then return end
    
        local box = render.get_player_bbox(local_player)
        if not box then return end
    
        render.outlined_rect(
            box.left,
            box.top,
            box.right - box.left,
            box.bottom - box.top,
            color.new(255, 255, 255, 255)
        )
    
        render.text(
            font.font,
            text_align_center,
            false,
            false,
            box.left + ((box.right - box.left) / 2),
            box.top - 14,
            color.new(255, 255, 255, 255),
            "local player"
        )
    end
    
    client.register_callback("paint_traverse", on_paint)

# callbacks | Evvware API

For the complete documentation index, see [llms.txt](https://evvware.gitbook.io/evvware-api/llms.txt)
. This page is also available as [Markdown](https://evvware.gitbook.io/evvware-api/variables/callbacks.md)
.

Use these names with `client.register_callback(string event, function)`.

### Movement and prediction[](https://evvware.gitbook.io/evvware-api/variables/callbacks#movement-and-prediction)

*   `"pre_create_move"` — runs before prediction
    
*   `"create_move"` — runs after prediction
    

### Frame stages[](https://evvware.gitbook.io/evvware-api/variables/callbacks#frame-stages)

*   `"pre_frame_stage_notify"` — runs before the original call
    
*   `"frame_stage_notify"` — runs before the original call
    

### Events and rendering[](https://evvware.gitbook.io/evvware-api/variables/callbacks#events-and-rendering)

*   `"aim_fire"` — event callback with 2 arguments
    
*   `"game_event_listener"` — receives events such as `player_say`, `player_spawn`, `round_start`, `player_death`, `player_hurt`, and `bullet_impact`
    
*   `"paint_traverse"` — runs after the original call and after ESP
    
*   `"on_antiaim"` — runs after all cheat-side anti-aim checks
    
*   `"on_unload"` — runs before the script fully unloads
    

### Handler arguments[](https://evvware.gitbook.io/evvware-api/variables/callbacks#handler-arguments)

*   `"pre_create_move"` and `"create_move"` receive [`usercmd`](https://evvware.gitbook.io/evvware-api/variables/classes#usercmd-class)
    
*   `"pre_frame_stage_notify"` and `"frame_stage_notify"` receive [`client_frame_stage`](https://evvware.gitbook.io/evvware-api/variables/classes#client_frame_stage-enum)
    
*   `"game_event_listener"` receives [`game_event`](https://evvware.gitbook.io/evvware-api/variables/classes#game_event-class)
    
*   `"paint_traverse"` receives no arguments
    
*   `"on_antiaim"` receives [`usercmd`](https://evvware.gitbook.io/evvware-api/variables/classes#usercmd-class)
    

### Examples[](https://evvware.gitbook.io/evvware-api/variables/callbacks#examples)

Copy

    local function on_create_move(cmd)
        cmd.viewangles = vector3.new(0, 0, 0)
    end
    
    local function pre_create_move(cmd)
        cmd.forwardmove = 400
    end
    
    local function on_antiaim(cmd)
        if client.get_send_packet() then
            cmd.viewangles.y = 180 --send fake angle
        else
            cmd.viewangles.y = 270 --send real angle
        end
    end
    
    local function on_frame_stage(stage)
        if stage == client_frame_stage.FRAME_RENDER_START then
            -- do something
        end
    end
    
    local function on_game_event(event)
        local event_name = event:get_name()
        
        if event_name = "player_hurt" then
            -- do something
        end
    end
    
    local function on_paint()
        --render.*
    end
    
    local function on_unload()
        collectgarbage("collect")
    end
    
    --callback registration example
    client.register_callback("on_antiaim" --[[string]], on_antiaim --[[function]])

### Minimal callback setup

Most scripts start by registering one or more callbacks.

\`\`\`lua
local function on\_create\_move(cmd)
    if not cmd then return end
end

local function on\_paint()
end

client.register\_callback("create\_move", on\_create\_move)
client.register\_callback("paint\_traverse", on\_paint)
\`\`\`

### Print to console and show notifications

Use \`client\` for quick feedback while testing.

\`\`\`lua
client.print("hello from lua")
client.print("colored message", color.new(120, 255, 120, 255))

client.add\_notify(
    "script loaded",
    true,
    true,
    color.new(255, 255, 255, 255),
    4.0
)
\`\`\`

### Read the local player

Use \`entities\` to get the local player, then read values from \`cs\_player\`.

\`\`\`lua
local function on\_create\_move(cmd)
    local local\_player = entities.get\_local\_player()
    if not local\_player then return end
    if not local\_player:is\_alive() then return end

    local health = local\_player:get\_health()
    local origin = local\_player:origin()
    local velocity = local\_player:velocity()

    client.print("health: " .. tostring(health))
    client.print("origin: " .. tostring(origin))
    client.print("velocity: " .. tostring(velocity))
end
\`\`\`

### Read engine state

Use \`engine\` for map, connection, screen, and view data.

\`\`\`lua
local screen = engine.get\_screen\_size()
local view = engine.get\_view\_angles()

client.print("map: " .. tostring(engine.get\_map\_name()))
client.print("screen: " .. tostring(screen))
client.print("view: " .. tostring(view))
client.print("in game: " .. tostring(engine.is\_in\_game()))
\`\`\`

### Loop players

Use \`engine.get\_active\_players\_data()\` when you need cached player data.

\`\`\`lua
for \_, player in ipairs(engine.get\_active\_players\_data()) do
    client.print(
        "index: " .. tostring(player.index) ..
        " alive: " .. tostring(player.alive) ..
        " teammate: " .. tostring(player.teammate)
    )
end
\`\`\`

### Handle game events

Register the event name first, then handle it in \`game\_event\_listener\`.

\`\`\`lua
local function on\_game\_event(event)
    local name = tostring(event:get\_name())

    if name == "bullet\_impact" then
        client.print("bullet impact")
    end

    if name == "player\_hurt" then
        local userid = event:get\_int("userid")
        client.print("hurt userid: " .. tostring(userid))
    end
end

-- also, you can register more events
-- client.register\_event("bomb\_planted")
-- check out: 
-- wiki.alliedmods.net/Counter-Strike:\_Source\_Events
-- wiki.alliedmods.net/Generic\_Source\_Events
-- make sure game supports event you choosed

client.register\_callback("game\_event\_listener", on\_game\_event)
\`\`\`

### Draw on screen

Use \`render\` inside \`paint\_traverse\`.

\`\`\`lua
local tahoma = render.create\_font("Tahoma", 12, 400, 0, 0, 16)

local function on\_paint()
    render.filled\_rect(20, 20, 220, 60, color.new(20, 20, 20, 180))
    render.outlined\_rect(20, 20, 220, 60, color.new(255, 255, 255, 255))

    render.text(
        tahoma.font,
        text\_align\_left,
        false,
        false,
        30,
        40,
        color.new(255, 255, 255, 255),
        "Hello from render"
    )
end
\`\`\`

### Convert world to screen

Use \`world2screen\` before drawing world positions.

\`\`\`lua
local function on\_paint()
    local local\_player = entities.get\_local\_player()
    if not local\_player then return end

    local screen\_pos = render.world2screen(local\_player:origin())
    if screen\_pos.x == -1 and screen\_pos.y == -1 then return end

    render.filled\_rect(
        screen\_pos.x - 2,
        screen\_pos.y - 2,
        4,
        4,
        color.new(255, 0, 0, 255)
    )
end
\`\`\`

### Build a menu

Use \`create\_ui\_manager()\` and add widgets to a frame.

\`\`\`lua
local ui\_manager = create\_ui\_manager()
local frame = ui\_manager:add\_frame()

frame:show()
frame:text("Example script")
frame:jump()

local enabled = frame:add\_checkbox("Enable", false, function(value)
    client.print("enabled: " .. tostring(value))
end)

local size = frame:add\_slider\_int("Size", 10, 1, 50, function(value)
    client.print("size: " .. tostring(value))
end)
\`\`\`

### Read widget values

Most menu widgets return a \`menu\_widget\`.

\`\`\`lua
local function on\_paint()

    local box\_size = size:get\_int()
    render.filled\_rect(300, 200, box\_size, box\_size, color.new(255, 255, 255, 255))
end

### Work with vectors and colors

\`vector2\`, \`vector3\`, and \`color\` are used everywhere.

\`\`\`lua
local a = vector3.new(10, 0, 0)
local b = vector3.new(5, 5, 0)
local c = a + b

client.print("vector result: " .. tostring(c))

local white = color.new(255, 255, 255, 255)
client.print("color alpha: " .. tostring(white.a))
\`\`\`

### Trace a ray

Use \`game\_ray\`, \`game\_trace\`, and a trace filter.

\`\`\`lua
local function on\_create\_move(cmd)
    local local\_player = entities.get\_local\_player()
    if not local\_player then return end

    local start\_pos = local\_player:eye\_position()
    local end\_pos = start\_pos + vector3.new(100, 0, 0)

    local ray = game\_ray.new()
    ray:init(start\_pos, end\_pos)

    local trace = game\_trace.new()
    local filter = trace\_filter\_custom.new(function(ent, mask)
        return true
    end)

    engine.trace\_ray(ray, MASK\_SOLID, filter, trace)

    client.print("fraction: " .. tostring(trace.fraction))
end
\`\`\`

### Use hitbox helpers

Use \`client\` helpers when you need hitbox positions fast.

    client.print("box min: " .. tostring(box\_min))
    client.print("box max: " .. tostring(box\_max))

### Work with materials

Use \`materials.get\_game\_textures()\` to inspect textures and materials.

\`\`\`lua
for \_, texture in ipairs(materials.get\_game\_textures()) do
    local material = texture.material
    if material and not material:is\_error\_material() then
        client.print(tostring(material:get\_name()))
        break
    end
end
\`\`\`

### Change material flags

Only do this when you know which material you are editing.

\`\`\`lua
for \_, texture in ipairs(materials.get\_game\_textures()) do
    local material = texture.material
    if material and not material:is\_error\_material() then
        material:set\_material\_flag(MATERIAL\_VAR\_WIREFRAME, true)
        break
    end
end
\`\`\`

### Get the active weapon config path

Use the local player weapon with \`config.get\_current\_aimbot\_config\_path()\`.

\`\`\`lua
local function on\_create\_move(cmd)
    local local\_player = entities.get\_local\_player()
    if not local\_player then return end

    local weapon = local\_player:get\_active\_weapon()
    if not weapon then return end

    local path = config.get\_current\_aimbot\_config\_path(weapon)
    client.print("config path: " .. tostring(path))
end
\`\`\`

### Full starter example

This combines callbacks, player lookup, render, and menu state.

\`\`\`lua
local ui\_manager = create\_ui\_manager()
local frame = ui\_manager:add\_frame()
frame:show()

local enabled = frame:add\_checkbox("Draw local player box", true)
local font = render.create\_font("Tahoma", 12, 400, 0, 0, 16)

local function on\_paint()
    if not enabled:get\_bool() then return end

    local local\_player = entities.get\_local\_player()
    if not local\_player then return end

    local box = render.get\_player\_bbox(local\_player)
    if not box then return end

    render.outlined\_rect(
        box.left,
        box.top,
        box.right - box.left,
        box.bottom - box.top,
        color.new(255, 255, 255, 255)
    )

    render.text(
        font.font,
        text\_align\_center,
        false,
        false,
        box.left + ((box.right - box.left) / 2),
        box.top - 14,
        color.new(255, 255, 255, 255),
        "local player"
    )
end

client.register\_callback("paint\_traverse", on\_paint)
\`\`\`

# client

Client helpers for console output, callbacks, etc.

### Console and notifications

\* \`.print(string message)\` — prints a message to the game console
\* \`.print(string message, color col)\` — prints a colored message to the game console
\* \`.add\_notify(string message, bool console\_message, bool anims, color col, float survival\_time)\` — shows a notification

### Events and callbacks

\* \`.register\_event(string name, bool server\_side = false)\` — registers an event for \[\`game\_event\_listener\`\](/evvware-api/variables/callbacks.md)&#x20;
\* \`.register\_callback(string event, function)\` — registers a Lua callback

### ESP flags

Checkout \[esp\\\_flag\](/evvware-api/variables/classes.md#esp\_flag-structure) class

\* \`.add\_esp\_flag(cs\_player index, esp\_flag esp\_flag\_data)\` — adds a custom ESP flag
\* \`.remove\_esp\_flag(cs\_player index, esp\_flag esp\_flag\_key)\` — removes a custom ESP flag

### Network state

\* \`.get\_send\_packet()\` returns \`bool\`
\* \`.set\_send\_packet(bool packet)\` — changes the send-packet state

### Hitbox helpers

\* \`.get\_hitbox\_box(cs\_player player, int id)\` returns 2 \[\`vector3\`\](/evvware-api/variables/classes.md#vector3-class) values — hitbox min and hitbox max
\* \`.get\_hitbox\_position(cs\_player player, int id)\` returns \[\`vector3\`\](/evvware-api/variables/classes.md#vector3-class)

### Version

\* \`.get\_cheat\_version()\` returns \`string\`

# callbacks

Use these names with \`client.register\_callback(string event, function)\`.

### Movement and prediction

\* \`"pre\_create\_move"\` — runs before prediction
\* \`"create\_move"\` — runs after prediction

### Frame stages

\* \`"pre\_frame\_stage\_notify"\` — runs before the original call
\* \`"frame\_stage\_notify"\` — runs before the original call

### Events and rendering

\* \`"aim\_fire"\` — event callback with 2 arguments
\* \`"game\_event\_listener"\` — receives events such as \`player\_say\`, \`player\_spawn\`, \`round\_start\`, \`player\_death\`, \`player\_hurt\`, and \`bullet\_impact\`
\* \`"paint\_traverse"\` — runs after the original call and after ESP
\* \`"on\_antiaim"\` — runs after all cheat-side anti-aim checks
\* \`"on\_unload"\` — runs before the script fully unloads

### Handler arguments

\* \`"pre\_create\_move"\` and \`"create\_move"\` receive \[\`usercmd\`\](/evvware-api/variables/classes.md#usercmd-class)
\* \`"pre\_frame\_stage\_notify"\` and \`"frame\_stage\_notify"\` receive \[\`client\_frame\_stage\`\](/evvware-api/variables/classes.md#client\_frame\_stage-enum)
\* \`"game\_event\_listener"\` receives \[\`game\_event\`\](/evvware-api/variables/classes.md#game\_event-class)
\* \`"paint\_traverse"\` receives no arguments
\* \`"on\_antiaim"\` receives \[\`usercmd\`\](/evvware-api/variables/classes.md#usercmd-class)

### Examples

\`\`\`lua
local function on\_create\_move(cmd)
    cmd.viewangles = vector3.new(0, 0, 0)
end

local function pre\_create\_move(cmd)
    cmd.forwardmove = 400
end

local function on\_antiaim(cmd)
    if client.get\_send\_packet() then
        cmd.viewangles.y = 180 --send fake angle
    else
        cmd.viewangles.y = 270 --send real angle
    end
end

local function on\_frame\_stage(stage)
    if stage == client\_frame\_stage.FRAME\_RENDER\_START then
        -- do something
    end
end

local function on\_game\_event(event)
    local event\_name = event:get\_name()
    
    if event\_name = "player\_hurt" then
        -- do something
    end
end

local function on\_paint()
    --render.\*
end

local function on\_unload()
    collectgarbage("collect")
end

--callback registration example
client.register\_callback("on\_antiaim" --\[\[string\]\], on\_antiaim --\[\[function\]\])

# materials

Material and texture access helpers.

### Functions

\* \`.get\_game\_textures()\` returns a table of \[\`texture\_structure\`\](#texture\_structure-struct)

### texture\\\_structure : struct

#### Values

\* \`.material\` returns \[\`imaterial\`\](#imaterial-class)
\* \`.group\` returns the texture group name as \`string\`

### imaterial : class

#### Functions

\* \`:set\_material\_flag(int flag, bool state)\` — sets a \[\`material\_flags\`\](#material\_flags-enum) value
\* \`:is\_error\_material()\` returns \`bool\`
\* \`:get\_num\_passes()\` — extra sanity check
\* \`:get\_texture\_memory\_bytes()\` returns the texture size in bytes
\* \`:get\_name()\` returns the material name as \`string\`
\* \`:get\_texture\_group\_name()\` returns the material group name as \`string\`
\* \`:increment\_reference\_count()\` — increments the material reference count
\* \`:alpha\_modulate(float alpha)\` — sets alpha modulation in the range \`0.0\` to \`1.0\`
\* \`:color\_modulate(float red, float green, float blue)\` — sets color modulation in the range \`0.0\` to \`1.0\`
